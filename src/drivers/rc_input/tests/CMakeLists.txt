cmake_minimum_required(VERSION 3.16)
project(sbus_pulse_tests LANGUAGES CXX)
enable_testing()
add_executable(sbus_pulse_test SBusPulseDecoderTest.cpp)
target_compile_features(sbus_pulse_test PRIVATE cxx_std_17)
target_compile_options(sbus_pulse_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
option(SBUS_SANITIZERS "Enable address/undefined behavior sanitizers" OFF)
if(SBUS_SANITIZERS)
	target_compile_options(sbus_pulse_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
	target_link_options(sbus_pulse_test PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME sbus_pulse_decoder COMMAND sbus_pulse_test)
